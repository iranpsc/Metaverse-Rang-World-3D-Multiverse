Shader "Meta/#SCRIPTNAME#"
{
    Properties
    {
        _MainTex ("Albedo (RGB)", 2D) = "white" {}
        _Color ("Tint Color", Color) = (1,1,1,1)
        _Metallic ("Metallic", Range(0,1)) = 0.0
        _Smoothness ("Smoothness", Range(0,1)) = 0.5
        _NormalMap ("Normal Map", 2D) = "bump" {}
        _EmissionColor ("Emission Color", Color) = (0,0,0,0)
    }

    SubShader
    {
        Tags { "RenderType"="Opaque" }
        LOD 200

        CGPROGRAM
        // دستوراتی که Surface Shader را تعریف می‌کنند
        #pragma surface surf Standard fullforwardshadows
        #pragma target 3.0

        sampler2D _MainTex;
        sampler2D _NormalMap;
        float4 _Color;
        float _Metallic;
        float _Smoothness;
        float4 _EmissionColor;

        struct Input
        {
            float2 uv_MainTex;
            float2 uv_NormalMap;
            float3 worldPos;
            float3 viewDir;
        };

        // تابع surf: هر پیکسل برای محاسبه رنگ نهایی به اینجا می‌آید
        void surf(Input IN, inout SurfaceOutputStandard o)
        {
            // آلبدو تکسچر و تیند
            float4 albedoSample = tex2D(_MainTex, IN.uv_MainTex) * _Color;
            o.Albedo = albedoSample.rgb;

            // نرمال مپ (اگر نداشتید حذف کنید)
            float3 bump = UnpackNormal(tex2D(_NormalMap, IN.uv_NormalMap));
            o.Normal = bump;

            // فلزی/صاف بودن
            o.Metallic = _Metallic;
            o.Smoothness = _Smoothness;

            // انتشار (emission)
            o.Emission = _EmissionColor.rgb * _EmissionColor.a;

            // آلفا (برای opaque اینجا 1 می‌گذاریم)
            o.Alpha = albedoSample.a;
        }
        ENDCG
    }

    FallBack "Diffuse"
}
